// Profile picture upload
$allowed = ['image/jpeg','image/png',
            'image/gif','image/webp'];
$maxSize = 2 * 1024 * 1024; // 2MB

if(in_array($_FILES['avatar']
   ['type'], $allowed) &&
   $_FILES['avatar']['size'] <= $maxSize){
  $ext = pathinfo(
    $_FILES['avatar']['name'],
    PATHINFO_EXTENSION
  );
  $name = uniqid().'.'.$ext;
  move_uploaded_file(
    $_FILES['avatar']['tmp_name'],
    'uploads/'.$name
  );
  // Update DB
  $stmt = $conn->prepare(
   "UPDATE users SET avatar=?
    WHERE id=?"
  );
  $stmt->bind_param('si',$name,$id);
  $stmt->execute();
}
